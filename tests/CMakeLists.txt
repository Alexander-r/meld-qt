# Test suite: ports of the original's Python tests, plus differential
# tests against reference data generated from the original implementation
# (see tools/generate_reference_data.py).

function(meld_add_test name)
    cmake_parse_arguments(ARG "" "" "SOURCES;LIBRARIES" ${ARGN})
    add_executable(${name} ${ARG_SOURCES} testutils.h)
    target_link_libraries(${name} PRIVATE Qt6::Test ${ARG_LIBRARIES})
    target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_definitions(${name} PRIVATE MELD_TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data")
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endfunction()
